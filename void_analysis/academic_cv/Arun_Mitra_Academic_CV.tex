\section{Research Experience}

\entryheader{Off-Campus Master's Research Thesis --- Cosmic Void Statistics \& Parameter Inference}{Sep 2025 -- Jun 2026}{External Advisors: Prof.\ Federico Marulli (Uni.\ Bologna / INAF), Dr.\ Simone Sartori (INAF-IASF Milano)}{Bologna, Italy / India}
\textit{BITS Pilani Co-Supervisors: Prof.\ Sashideep Gutti, Prof.\ Ram Chandra Murthy Kalluri}
\begin{itemize}
    \item \textbf{Excursion-Set Void Modeling on $N$-body Simulations:} Evaluated the Jennings et al.\ (2013) Volume-Conserving (Vdn) theoretical model on the $(1\,h^{-1}\mathrm{Gpc})^3$ Aletheia simulation suite across four mass-thresholded halo catalogues (100P, 75P, 50P, 30P; effective linear bias $b_{\mathrm{eff}} \in [0.86, 1.02]$).
    \item \textbf{Discovery of Linear Void Barrier Universality ($B_{\mathrm{slope}} \sim 0.9$):} Discovered that linear barrier scaling ($b_{\mathrm{void}} = B_{\mathrm{slope}} b_{\mathrm{eff}} + B_{\mathrm{offset}}$) matches between dynamical Optimal Transport reconstruction (\texttt{BitVF}) and Voronoi watershed (\texttt{REVOLVER}) to an unprecedented \textbf{$0.07\sigma$} ($B_{\mathrm{slope}}^{\mathrm{BitVF}} = 0.914 \pm 0.043$ vs.\ $B_{\mathrm{slope}}^{\mathrm{REVOLVER}} = 0.910 \pm 0.046$). Proved that linear barrier response to tracer bias is a universal property of excursion-set void cosmology rather than an algorithm-specific artifact.
    \item \textbf{Model-Finder Compatibility \& MCMC Bias Resolution:} Proved that the idealized spherical Vdn excursion-set model is structurally compatible with dynamical Lagrangian reconstruction (\texttt{BitVF}, $\chi^2/\nu \approx 0.90\text{--}1.67$), but fails for Voronoi watershed finders (\texttt{REVOLVER}, $\chi^2/\nu \approx 3.14\text{--}8.12$). Demonstrated that unmerged watershed basins over-partition large voids, creating a systematic residual slope tilt that drives Bayesian MCMC samplers toward unphysical cosmological parameters ($\Omega_m \sim 0.44$).
    \item \textbf{Bayesian Parameter Inference Pipeline:} Developed complete Python/C++ MCMC sampling framework using \texttt{CosmoBolognaLib} (CBL) and \texttt{emcee}. Calibrated informative Gaussian priors from simulation mocks, successfully recovering primordial amplitude $\sigma_8$ to \textbf{$1.2\%$ precision} ($\bar{\sigma}_8 = 0.8208 \pm 0.009$ for BitVF vs.\ fiducial $0.8110$).
    \item \textbf{Robustness Audits \& Volume Conservation Prerequisite:} Constructed 64-region spatial Jackknife covariance matrices debiased via the Anderson-Hartlap factor. Demonstrated that Vdn modeling mathematically requires non-overlapping spherical voids (\texttt{TOUCH}), where concentric center overlaps (\texttt{CENTER}) double-count volume and artificially inflate $\chi^2/\nu$ to $\sim 25$.
\end{itemize}

\entryheader{Research Projects --- Primordial Black Holes (PBHs) \& Early Universe Cosmology}{2024 -- 2025}{Department of Physics, BITS Pilani}{Hyderabad, India}

Studied the critical collapse overdensity threshold $\delta_c$ and mass fraction $\beta(M)$.
    \item \textbf{Dark Matter Fraction \& Observational Windows:} Explored current observational limits on $f_{\mathrm{PBH}} = \Omega_{\mathrm{PBH}}/\Omega_{\mathrm{DM}}$ across different mass regimes, including Hawking evaporation bounds (low mass), gravitational microlensing (intermediate mass), and accretion limits from CMB anisotropies.
    \item \textbf{Linear Perturbation Theory \& Growth Factors:} Analyzed the evolution of scalar cosmological perturbations, Jeans mass scales, and power spectrum calculations using numerical Boltzmann solvers (CAMB).
\end{itemize}
